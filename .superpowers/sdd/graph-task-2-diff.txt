+"""Graph memory read endpoints: POST /graph/search and GET /graph/get_all.
+
+Scope contract (Task 1 review requirement):
+- `_resolve_scope` always returns a dict containing a `scope` key:
+  "project", "user", or "none".
+- Project scope always carries a non-empty project_id (resolved from
+  git_remote when needed).
+- Personal scope pins user_id to the caller (non-admin) or to the
+  admin-provided user_id; a normal user passing another user's id → 403.
+- Admin with neither project nor user_id → scope "none" → the router
+  returns an empty result without touching the graph (GraphMemory itself
+  also guards unscoped reads — see graph_memory.search/get_all).
+"""
+
+from typing import Optional
+
+from auth import verify_auth
+from fastapi import APIRouter, Depends, HTTPException
+from gitmatch import resolve_project_id
+from pydantic import BaseModel, Field
+from server_state import get_graph_memory
+
+router = APIRouter(prefix="/graph", tags=["graph"])
+
+
+class GraphSearchRequest(BaseModel):
+    query: str
+    project_id: Optional[str] = None
+    git_remote: Optional[str] = None
+    user_id: Optional[str] = None  # admin only
+    limit: int = Field(25, ge=1, le=1000)
+
+
+class GraphGetAllRequest(BaseModel):
+    project_id: Optional[str] = None
+    git_remote: Optional[str] = None
+    user_id: Optional[str] = None  # admin only
+    limit: int = Field(100, ge=1, le=1000)
+

+def _resolve_scope(user, project_id, git_remote, user_id, db):
+    """Return scope filters (always containing `scope`) or raise HTTPException.
+
+    Personal scope is pinned to the caller: a non-admin can never widen the
+    read to another user (403). An admin with no project and no user_id gets
+    scope "none" — an intentional empty result, not an error, so the UI can
+    render an empty graph instead of surfacing a 400.
+    """
+    if project_id is None and git_remote:
+        project_id = resolve_project_id(git_remote, db)
+    is_admin = user is None or getattr(user, "role", None) == "admin"
+    if not is_admin:
+        if user_id and str(user_id) != str(user.id):
+            raise HTTPException(status_code=403, detail="Cannot access graph of another user_id.")
+        return {
+            "user_id": str(user.id),
+            "project_id": project_id,
+            "scope": "project" if project_id else "user",
+        }
+    if user_id:
+        return {"user_id": str(user_id), "project_id": project_id, "scope": "user"}
+    if project_id:
+        return {"project_id": project_id, "scope": "project"}
+    return {"scope": "none"}
+
+
+@router.post("/search")
+def graph_search(body: GraphSearchRequest, _auth=Depends(verify_auth), db=None):
+    from db import SessionLocal
+
+    session = SessionLocal() if db is None else db
+    try:
+        filters = _resolve_scope(_auth, body.project_id, body.git_remote, body.user_id, session)
+        if filters.get("scope") == "none":
+            return {"relations": []}
+        graph = get_graph_memory()
+        return {"relations": graph.search(body.query, filters, limit=body.limit)}
+    finally:
+        if db is None:
+            session.close()
+
+
