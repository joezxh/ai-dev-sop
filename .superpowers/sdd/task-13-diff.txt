+"use client";
+
+import React, { useEffect, useState } from "react";
+import { Download, FileDown } from "lucide-react";
+import { Button } from "@/components/ui/button";
+import { Input } from "@/components/ui/input";
+import { Label } from "@/components/ui/label";
+import { Card } from "@/components/ui/card";
+import {
+  Select,
+  SelectContent,
+  SelectItem,
+  SelectTrigger,
+  SelectValue,
+} from "@/components/ui/select";
+import { toast } from "@/components/ui/use-toast";
+import { TableSkeleton } from "@/components/shared/table-skeleton";
+import { EmptyState } from "@/components/self-hosted/empty-state";
+import { api } from "@/utils/api";
+import { EXPORT_ENDPOINTS, PROJECT_ENDPOINTS } from "@/utils/api-endpoints";
+import { Project } from "@/types/api";
+import { useTranslation } from "@/i18n";
+
+export default function MemoryExportsPage() {
+  const { t } = useTranslation();
+  const [projects, setProjects] = useState<Project[]>([]);
+  const [loading, setLoading] = useState(true);
+  const [error, setError] = useState<string | null>(null);
+
+  const [projectId, setProjectId] = useState("all");
+  const [userId, setUserId] = useState("");
+  const [start, setStart] = useState("");
+  const [end, setEnd] = useState("");
+  const [exporting, setExporting] = useState(false);
+
+  async function load() {
+    setLoading(true);
+    try {
+      const res = await api.get(PROJECT_ENDPOINTS.BASE);
+      setProjects(res.data ?? []);
+      setError(null);
+    } catch (e: any) {
+      setError(e?.response?.data?.detail ?? e?.message ?? "Failed to load projects");
+    } finally {
+      setLoading(false);
+    }
+  }
+
+  useEffect(() => {
+    load();
+    // eslint-disable-next-line react-hooks/exhaustive-deps
+  }, []);
+
+  async function exportMemories() {
+    const params: Record<string, string> = { format: "jsonl" };
+    if (projectId && projectId !== "all") params.project_id = projectId;
+    if (userId.trim()) params.user_id = userId.trim();
+    if (start) params.start = start;
+    if (end) params.end = end;
+
+    setExporting(true);
+    try {
+      // Blob download instead of window.open: the endpoint requires an admin
+      // JWT, which window.open cannot attach in authenticated mode.
+      const res = await api.get(EXPORT_ENDPOINTS.MEMORIES, {
+        params,
+        responseType: "blob",
+      });
+      const url = URL.createObjectURL(res.data as Blob);
+      const a = document.createElement("a");
+      a.href = url;
+      a.download = "memories-export.jsonl";
+      a.click();
+      URL.revokeObjectURL(url);
+      setError(null);
+      toast({ title: "Export downloaded", variant: "success" });
+    } catch (e: any) {
+      const detail =
+        typeof e === "string"
+          ? e
+          : (e?.response?.data?.detail ?? e?.message ?? "Export failed");
+      setError(detail);
+      toast({ title: detail, variant: "warning" });
+    } finally {
+      setExporting(false);
+    }
+  }
+
+  return (
+    <div className="p-6 space-y-6">
+      <div>
+        <h1 className="text-2xl font-semibold">{t("exports.title")}</h1>
+        <p className="text-memText-secondary text-sm">{t("exports.desc")}</p>
+      </div>
+      {error && <div className="text-red-500 text-sm">{error}</div>}
+
+      {loading ? (
+        <TableSkeleton rows={3} columns={3} />
+      ) : (
+        <Card className="border-memBorder-primary p-6 max-w-2xl space-y-4">
+          <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
+            <div className="space-y-2">
+              <Label htmlFor="export-project">{t("tenant.projects")}</Label>
+              <Select value={projectId} onValueChange={setProjectId}>
+                <SelectTrigger id="export-project" className="w-full">
+                  <SelectValue placeholder={t("exports.allProjects")} />
+                </SelectTrigger>
+                <SelectContent>
+                  <SelectItem value="all">{t("exports.allProjects")}</SelectItem>
+                  {projects.map((p) => (
+                    <SelectItem key={p.id} value={p.project_id}>
+                      {p.name || p.project_id}
+                    </SelectItem>
+                  ))}
+                </SelectContent>
+              </Select>
+            </div>
+            <div className="space-y-2">
+              <Label htmlFor="export-user">{t("exports.userId")}</Label>
+              <Input
+                id="export-user"
+                placeholder={t("exports.userIdPlaceholder")}
+                value={userId}
+                onChange={(e) => setUserId(e.target.value)}
+              />
+            </div>
+            <div className="space-y-2">
+              <Label htmlFor="export-start">{t("exports.start")}</Label>
+              <Input
+                id="export-start"
+                type="date"
+                value={start}
+                onChange={(e) => setStart(e.target.value)}
+              />
+            </div>
+            <div className="space-y-2">
+              <Label htmlFor="export-end">{t("exports.end")}</Label>
+              <Input
+                id="export-end"
+                type="date"
+                value={end}
+                onChange={(e) => setEnd(e.target.value)}
+              />
+            </div>
+          </div>
+
+          <div className="flex items-center gap-3 pt-2">
+            <Button onClick={exportMemories} disabled={exporting}>
+              {exporting ? (
+                <FileDown className="size-4 mr-1 animate-pulse" />
+              ) : (
+                <Download className="size-4 mr-1" />
+              )}
+              {t("exports.button")}
+            </Button>
+            <span className="text-xs text-memText-secondary">
+              {t("exports.formatHint")}
+            </span>
+          </div>
+        </Card>
+      )}
+
+      {!loading && projects.length === 0 && (
+        <EmptyState
+          title={t("exports.noProjectsTitle")}
+          description={t("exports.noProjectsDesc")}
+        />
+      )}
+    </div>
+  );
+}
