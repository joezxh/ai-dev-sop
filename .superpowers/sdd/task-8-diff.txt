+"use client";
+
+import React, { useState } from "react";
+import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
+import Link from "next/link";
+
+// MCP 服务：mem0/server/mcp_server.py —— FastMCP("mem0-local")，streamable-http，
+// 监听 MEM0_MCP_PORT（默认 8080），挂载路径 /mcp。
+const MCP_JSON = `{
+  "mcpServers": {
+    "mem0-local": {
+      "url": "http://<your-server>:8080/mcp"
+    }
+  }
+}`;
+
+// 工具入参以 mcp_server.py 源码为准：
+//   api_key     —— 你的个人 API Key（/dashboard/api-keys 创建）
+//   git_remote  —— 本地工程 `git remote -v` 输出的地址，服务端自动解析 project_id
+const MCP_TOOLS = [
+  { name: "add_memory", args: "text, api_key, git_remote?, project_id?, metadata?" },
+  { name: "search_memories", args: "query, api_key, git_remote?, project_id?, top_k?" },
+  { name: "get_memories", args: "api_key, git_remote?, project_id?, limit?" },
+  { name: "match_project", args: "git_remote, api_key?" },
+];
+
+// REST：mem0-api，容器内 8000 / 宿主机 8888；鉴权 X-API-Key。
+const PYTHON = `import requests
+
+BASE = "http://<your-server>:8888"          # mem0-api REST 端口
+KEY = "YOUR_API_KEY"                         # /dashboard/api-keys 创建
+GIT_REMOTE = "git@github.com:org/repo.git"   # 本地 git remote -v
+
+# 1) 添加记忆（git_remote 自动解析 project_id 并写入 metadata）
+r = requests.post(
+    f"{BASE}/memories",
+    headers={"X-API-Key": KEY},
+    json={
+        "messages": [{"role": "user", "content": "本项目使用 FastAPI + pgvector"}],
+        "git_remote": GIT_REMOTE,
+    },
+)
+print(r.status_code, r.json())
+
+# 2) 检索记忆（带 git_remote 即检索该项目的共享池，跨用户）
+r = requests.post(
+    f"{BASE}/search",
+    headers={"X-API-Key": KEY},
+    json={"query": "本项目用什么数据库？", "git_remote": GIT_REMOTE},
+)
+print(r.json())`;
+
+const CURL = `# 1) 添加记忆
+curl -X POST http://<your-server>:8888/memories \\
+  -H "X-API-Key: YOUR_API_KEY" \\
+  -H "Content-Type: application/json" \\
+  -d '{"messages":[{"role":"user","content":"我喜欢深色主题"}],"git_remote":"git@github.com:org/repo.git"}'
+
+# 2) 检索记忆（带 git_remote 即检索该项目的共享池，跨用户）
+curl -X POST http://<your-server>:8888/search \\
+  -H "X-API-Key: YOUR_API_KEY" \\
+  -H "Content-Type: application/json" \\
+  -d '{"query":"主题偏好","git_remote":"git@github.com:org/repo.git"}'`;
+
+const TABS = [
+  { id: "mcp", label: "MCP（CodeBuddy / Qoder）", code: MCP_JSON },
+  { id: "python", label: "Python", code: PYTHON },
+  { id: "curl", label: "cURL", code: CURL },
+] as const;
+
+export default function GetStartedPage() {
+  const [tab, setTab] = useState<(typeof TABS)[number]["id"]>("mcp");
+  const active = TABS.find((t) => t.id === tab)!;
+
+  return (
+    <div className="space-y-6">
+      <h1 className="text-xl font-semibold font-fustat">Get Started</h1>
+
+      <Card className="border-memBorder-primary">
+        <CardHeader>
+          <CardTitle className="text-sm">Step 1 · 创建 API Key</CardTitle>
+        </CardHeader>
+        <CardContent className="text-sm">
+          在{" "}
+          <Link className="underline" href="/dashboard/api-keys">
+            API Keys
+          </Link>{" "}
+          页创建个人密钥（仅创建时可见）。所有接入方式都通过{" "}
+          <code>X-API-Key</code> 请求头识别你的身份。
+        </CardContent>
+      </Card>
+
+      <Card className="border-memBorder-primary">
+        <CardHeader>
+          <CardTitle className="text-sm">Step 2 · 选择接入方式</CardTitle>
+        </CardHeader>
+        <CardContent className="space-y-3">
+          <div className="flex flex-wrap gap-2">
+            {TABS.map((t) => (
+              <button
+                key={t.id}
+                onClick={() => setTab(t.id)}
+                className={`px-3 py-1.5 text-sm rounded border ${
+                  t.id === tab
+                    ? "bg-surface-default-brand text-onSurface-default-primary border-memBorder-primary"
+                    : "border-memBorder-primary hover:bg-surface-default-secondary"
+                }`}
+              >
+                {t.label}
+              </button>
+            ))}
+          </div>
+          <pre className="text-xs bg-surface-default-secondary p-3 rounded font-mono overflow-x-auto max-h-96">
+            {active.code}
+          </pre>
+          {tab === "mcp" && (
+            <div className="text-xs text-onSurface-default-secondary space-y-1">
+              <p>
+                配置文件填入上述 JSON 即可（transport：
+                <code>streamable-http</code>，默认端口 8080）。可用工具：
+              </p>
+              <ul className="space-y-0.5">
+                {MCP_TOOLS.map((tool) => (
+                  <li key={tool.name}>
+                    <code>{tool.name}</code>（{tool.args}）
+                  </li>
+                ))}
+              </ul>
+              <p>
+                其中 <code>api_key</code> 为 Step 1 创建的密钥，
+                <code>git_remote</code> 为本地 <code>git remote -v</code>{" "}
+                输出的地址，服务端自动匹配项目。
+              </p>
+            </div>
+          )}
+          {tab !== "mcp" && (
+            <p className="text-xs text-onSurface-default-secondary">
+              把 <code>&lt;your-server&gt;</code> 换成部署 mem0-api 的主机地址，
+              <code>YOUR_API_KEY</code> 换成 Step 1 创建的密钥，
+              <code>git_remote</code> 换成你本地工程的 remote 地址。
+            </p>
+          )}
+        </CardContent>
+      </Card>
+
+      <Card className="border-memBorder-primary">
+        <CardHeader>
+          <CardTitle className="text-sm">Step 3 · 验证记忆写入</CardTitle>
+        </CardHeader>
+        <CardContent className="text-sm">
+          运行上方“添加记忆”代码，然后到{" "}
+          <Link className="underline" href="/dashboard/memories">
+            Memories
+          </Link>{" "}
+          查看（metadata 中应包含 <code>git_remote</code> 与解析出的{" "}
+          <code>project_id</code>）。
+        </CardContent>
+      </Card>
+
+      <Card className="border-memBorder-primary">
+        <CardHeader>
+          <CardTitle className="text-sm">Step 4 · 检索与共享池</CardTitle>
+        </CardHeader>
+        <CardContent className="text-sm">
+          检索默认只返回你自己的记忆；带 <code>git_remote</code>/
+          <code>project_id</code> 的检索/拉取会返回该项目的共享池（跨用户）。
+          可先在{" "}
+          <Link className="underline" href="/dashboard/playground">
+            Playground
+          </Link>{" "}
+          无代码试玩。
+        </CardContent>
+      </Card>
+    </div>
+  );
+}
